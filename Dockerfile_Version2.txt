# -------------------------------
# Builder Stage: Composer & Node for Assets + Vendor Caching
# -------------------------------
FROM composer:2.7 AS build

# Install Node.js (LTS)
RUN apk add --no-cache nodejs npm

WORKDIR /app

# Copy composer manifests first for better layer caching
COPY composer.json composer.lock ./

# Install PHP dependencies for composer (simulate runtime platform)
ENV PHP_EXTENSIONS="bcmath ctype fileinfo json mbstring openssl pdo pdo_mysql tokenizer xml zip exif pcntl intl"
RUN composer config --global platform.php 8.2

# Install Composer dependencies (no dev, optimized for production, skip scripts)
RUN composer install --no-dev --prefer-dist --optimize-autoloader --no-scripts

# Copy package manifests for Node
COPY package.json package-lock.json* npm-shrinkwrap.json* ./

# Install Vite build (if present)
RUN if [ -f package.json ]; then npm ci --omit=dev && npm run build; fi

# Copy over rest of the app for asset compilation, public build, etc.
COPY . .

# -------------------------------
# Production Stage: Nginx + PHP-FPM + Supervisor, minimal Alpine image
# -------------------------------
FROM php:8.2-fpm-alpine AS app

LABEL maintainer="YourName <your@email.com>"

# Install OS & PHP extensions, Nginx, Supervisor, & tools
RUN apk add --no-cache --virtual .build-deps $PHPIZE_DEPS icu-dev oniguruma-dev libxml2-dev libzip-dev zlib-dev \
    && apk add --no-cache nginx supervisor bash tzdata curl git icu-data-full \
    && docker-php-ext-install bcmath ctype fileinfo intl mbstring pcntl pdo pdo_mysql tokenizer xml zip exif opcache \
    && apk del .build-deps

# Set recommended PHP.ini settings
COPY --from=build /usr/local/etc/php/conf.d /usr/local/etc/php/conf.d

# Copy the app files, vendors, and built assets from the build stage
WORKDIR /var/www/html
COPY --from=build /app ./

# Copy configs
COPY docker/nginx.conf /etc/nginx/nginx.conf
COPY docker/supervisord.conf /etc/supervisord.conf
COPY docker/docker-entrypoint.sh /usr/local/bin/docker-entrypoint.sh

RUN chmod +x /usr/local/bin/docker-entrypoint.sh

# Permissions
RUN chown -R www-data:www-data /var/www/html \
    && chmod -R 775 storage bootstrap/cache

# Clean npm & temp files
RUN rm -rf node_modules /root/.npm /tmp/* /var/cache/apk/*

# Healthcheck (requires curl)
HEALTHCHECK --interval=30s --timeout=5s --start-period=10s --retries=3 \
  CMD curl -f http://localhost/ || exit 1

# Expose port 80 to the outside world
EXPOSE 80

ENTRYPOINT ["/usr/local/bin/docker-entrypoint.sh"]
CMD ["/usr/bin/supervisord", "-c", "/etc/supervisord.conf"]

# --------- End Dockerfile -----------
